var require;
window.exports = {};
window.module = {exports: null};

(function (){
    const nodeMap = new Map();
    const scriptPaths = [];

    var get = function(module, path){
        if(scriptPaths.length > 0){
            return NWJSApi.getScriptFile(path, module, scriptPaths[scriptPaths.length - 1]);
        }
        return NWJSApi.getScriptFile(path, module, "undefined");
    }
    
    var getBase = function(){
        if(document.currentScript == null){
            return '';
        }

        var src = document.currentScript.getAttribute('path');
        if(!src)
            src = document.currentScript.src;
            
        if(src.includes('/')){
            var segments = src.split('/');
            segments.pop();
            return segments.join('/');
        } else {
            return '';
        }
    }

    var genScriptId = function(module){
        var src;

        if(document.currentScript == null){
            src = "";
        } else {
            src = document.currentScript.src;
        }

        return `${src}:${module}`
    }

    var getScriptId = function(module){
        if(document.currentScript == null){
            return '';
        }

        var id = document.currentScript.getAttribute('script-id');

        if(id){
            return id;
        } else {
            return "";
        }
    }
    
    var normalizePath = function(path){
        var segments = path.split('/');
        var paths = [];
        
        segments.forEach(segment => {
            if(segment == '.'){
            } else if ((segment == '..') && (paths.length > 0)){
                paths.pop();
            } else {
                paths.push(segment);
            }
        });
        
        return paths.join('/');
    }
    
    var loadFromModules = function(module){
        var packageJson = get(module, `${module}/package.json`);
        
        if(packageJson){
            var main = JSON.parse(packageJson).main;
            
            if(!main){
                main = 'index.js';
            }
            
            var scriptPath = normalizePath(`${module}/${main}`);
            
            if(scriptPath.endsWith('.json')){
                scriptPath = '.'+scriptPath;
            
                var data = get(module, scriptPath);
                if(data){
                    return {path: scriptPath, script: data, type: 'json', module: null};
                }
            }
            
            scriptPath = './'+scriptPath;
            
            var data = get(module, scriptPath);

            if(data){
                return {path: scriptPath, script: data, type: 'js', module: module};
            }
        }
    }
    
    var loadFromScript = function(module){
        var modulePath = normalizePath(module);
        var scriptPath = getBase()+'/'+module;

        scriptPath = normalizePath(scriptPath);

        var data = get(module, scriptPath);
        if(data){
            return {path: scriptPath, script: data, type: 'js', module: null};
        }

        data = get(module, modulePath);
        if(data){
            return {path: modulePath, script: data, type: 'js', module: null};
        }
    }

    var loadFromModuleScript = function(module){
        var scriptPath = `./${module}`
        
        scriptPath = normalizePath(scriptPath);
        scriptPath = './'+scriptPath;

        var data = get(module, scriptPath);
        if(data){
            return {path: scriptPath, script: data, type: 'js', module: null};
        }
    } 
    
    var loadModule = function(module){
        var fromScript = loadFromScript(module);
        
        if(fromScript){
            return fromScript;
        } else {
            var fromModules = loadFromModules(module);
            
            if(fromModules){
                return fromModules;
            } else {
                var scriptFromModules = loadFromModuleScript(module);
                if(scriptFromModules){
                    return scriptFromModules;
                }
            }
        }
    }

    require = function (module){
        window.exports = {};
        window.module.exports = null

        var scriptId = genScriptId(module);

        if(nodeMap.has(scriptId)){
            if(typeof nodeMap.get(scriptId) === 'function'){
                return nodeMap.get(scriptId).bind({});
            } else {
                return Object.assign({}, nodeMap.get(scriptId));
            }
        }
        
        var data = loadModule(module);
        
        if(data == null){
            throw new Error(`${module} could not be found in module paths.`);
        }

        if(data.type == 'json'){
            return JSON.parse(data.script);
        }

        if(scriptPaths[scriptPaths.length - 1] && !data.module){
            scriptPaths.push(NWJSApi.getChildPath(scriptPaths[scriptPaths.length - 1], data.path))
        } else {
            scriptPaths.push(data.path)
        }

        eval?.(data.script);

        var object = null;
        if(window.module.exports == null){
            object = window.exports;
        } else {
            object = window.module.exports;
        }
        
        if(object == null){
            console.log(`${module} exports a null object`);
            return null;
        }

        scriptPaths.pop()
        try{
            if(typeof object === 'function'){
                nodeMap.set(scriptId, object.bind({}));
                return object.bind({});
            } else {
                nodeMap.set(scriptId, Object.assign({}, object));
                return Object.assign({}, object);
            }
        } catch(e) {
            return object;
        }
        
    }
    
    window.exports = {}
    
    window.module = {
        exports: null
    }

})()
