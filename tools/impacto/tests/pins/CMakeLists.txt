cmake_minimum_required(VERSION 3.22)
project(impacto LANGUAGES NONE)
include(FetchContent)
# Stand-ins for upstream's later declarations. Do not populate/download them.
FetchContent_Declare(libatrac9 GIT_REPOSITORY https://invalid.test/unpinned GIT_TAG master)
FetchContent_Declare(ImGui GIT_REPOSITORY https://invalid.test/unpinned GIT_TAG moving-tag)
FetchContent_Declare(magic_enum GIT_REPOSITORY https://invalid.test/unpinned GIT_TAG master)
file(READ "${TYRANOR_IMPACTO_SOURCE_LOCK}" lock)
foreach(dep libatrac9 imgui magic_enum)
    get_property(details GLOBAL PROPERTY "_FetchContent_${dep}_savedDetails")
    string(JSON revision GET "${lock}" "${dep}" revision)
    string(JSON repository GET "${lock}" "${dep}" url)
    list(FIND details "${revision}" pin)
    list(FIND details "${repository}" url)
    if(pin LESS 0 OR url LESS 0)
        message(FATAL_ERROR "Pinned source lost for ${dep}: ${details}")
    endif()
    message(STATUS "PASS: ${dep} retained its locked repository and revision")
endforeach()
get_property(atrac GLOBAL PROPERTY _FetchContent_libatrac9_savedDetails)
list(FIND atrac PATCH_COMMAND patch)
if(patch LESS 0)
    message(FATAL_ERROR "Upstream LibAtrac9 dependency recipe lost")
endif()
